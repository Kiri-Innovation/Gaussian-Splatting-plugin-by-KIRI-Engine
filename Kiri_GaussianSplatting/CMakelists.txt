# === shader gen ===
find_package(Python3 COMPONENTS Interpreter REQUIRED)

set(SHADER_GEN_H   ${CMAKE_CURRENT_SOURCE_DIR}/include/generated/shader_autogen.h)
set(SHADER_FOLDER  ${CMAKE_CURRENT_SOURCE_DIR}/shader)

file(GLOB_RECURSE SHADER_SRC_FILES
    CONFIGURE_DEPENDS
    ${SHADER_FOLDER}/*.glsl
    ${SHADER_FOLDER}/*.frag
    ${SHADER_FOLDER}/*.vert
    ${SHADER_FOLDER}/*.metal
	 ${SHADER_FOLDER}/*.comp
)

add_custom_command(
    OUTPUT  ${SHADER_GEN_H}
    DEPENDS ${SHADER_SRC_FILES} 
    COMMAND ${Python3_EXECUTABLE} 
		 	${SHADER_FOLDER}/CompileShader.py
            --shader_folder ${SHADER_FOLDER}
            --output ${SHADER_GEN_H}
    COMMENT "Compiling shader to C++ code..."
    VERBATIM
)
add_custom_target( GEN_SHADER
    DEPENDS ${SHADER_GEN_H}
)


# === shader gen ===


# ===== source + header  =====
file(GLOB_RECURSE SRC_FILES
    CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp
)

file(GLOB_RECURSE HEADER_FILES
    CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/include/*.h
    ${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp
	${SHADER_GEN_H}
)

add_library(${PROJECT_NAME} MODULE
    ${SRC_FILES}
    ${HEADER_FILES}
	${SHADER_SRC_FILES}
)

add_dependencies(${PROJECT_NAME} GEN_SHADER)


source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}/src"     PREFIX "src"     FILES ${SRC_FILES})
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}/include" PREFIX "include" FILES ${HEADER_FILES})
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}/shader"  PREFIX "shader"  FILES ${SHADER_SRC_FILES})

	message(STATUS "Platform: macOS")

    add_custom_command(
  		OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.rsrc
  		COMMAND ${AFX_REZ}

			${PIPL_INCLUDES}
      	-i ${AESDK_ROOT}/Headers

      	-i ${AESDK_ROOT}/Resources
  			-D __MACH__
  	)

	set_source_files_properties(
		${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.rsrc
		PROPERTIES
		MACOSX_PACKAGE_LOCATION Resources/
	)
    
    target_sources( ${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}.rsrc )

	set_target_properties( ${PROJECT_NAME} PROPERTIES PREFIX "" )
	set_target_properties( ${PROJECT_NAME} PROPERTIES SUFFIX "" )
	string( TIMESTAMP BUILD_YEAR "%Y" )
	set_target_properties( ${PROJECT_NAME} PROPERTIES 
    BUNDLE True
    MACOSX_BUNDLE_BUNDLE_NAME ${PROJECT_NAME}
    BUNDLE_EXTENSION "plugin"
    MACOSX_BUNDLE_INFO_STRING "${PROJECT_NAME}"
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.adobe.AfterEffects.${PROJECT_NAME}"
    MACOSX_BUNDLE_COPYRIGHT "MIT LICENSED ${BUILD_YEAR}."
  )

	# PkgInfo
	add_custom_command(
		OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/PkgInfo
		COMMAND echo "eFKTFXTC" > ${CMAKE_CURRENT_BINARY_DIR}/PkgInfo
	)

	set_source_files_properties(
		${CMAKE_CURRENT_BINARY_DIR}/PkgInfo
		PROPERTIES
		MACOSX_PACKAGE_LOCATION ""
	)

	target_sources( ${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/PkgInfo )

endif()
